import { useEffect, useMemo, useState } from 'react';
import { Bookmark as BookmarkIcon, Search, Trash2, X, Folder } from 'lucide-react';
import { useBookmarksStore, useTabStore } from '../store';
import type { BookmarkItem } from '../store/bookmarksStore';

export function BookmarksPage() {
  const { items, hydrated, hydrate, removeBookmark } = useBookmarksStore();
  const { getActiveTab, navigate, addTab } = useTabStore();
  const [query, setQuery] = useState('');

  useEffect(() => {
    if (!hydrated) hydrate();
  }, [hydrated, hydrate]);

  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return items;
    return items.filter(
      (b) => b.title.toLowerCase().includes(q) || b.url.toLowerCase().includes(q)
    );
  }, [items, query]);

  const grouped = useMemo(() => {
    const groups = new Map<string, BookmarkItem[]>();
    for (const b of filtered) {
      const key = b.folder ?? 'Unsorted';
      const bucket = groups.get(key);
      if (bucket) bucket.push(b);
      else groups.set(key, [b]);
    }
    return Array.from(groups.entries()).sort(([a], [b]) => a.localeCompare(b));
  }, [filtered]);

  const openUrl = (url: string, title: string) => {
    const activeTab = getActiveTab();
    if (activeTab) navigate(activeTab.id, url, title);
    else addTab(url);
  };

  return (
    <div className="flex-1 bg-[#0d0e10] p-8 overflow-y-auto">
      <div className="max-w-3xl mx-auto space-y-6 animate-in fade-in duration-300">
        <div className="flex items-center gap-3 border-b border-[#252830] pb-5">
          <div className="w-10 h-10 rounded-xl bg-amber-500/10 border border-amber-500/20 flex items-center justify-center text-amber-400">
            <BookmarkIcon className="w-5 h-5" />
          </div>
          <div>
            <h1 className="text-xl font-bold text-white tracking-tight">Bookmarks</h1>
            <p className="text-xs text-zinc-400">
              Saved pages, persisted locally. Star a page from the address bar to add one.
            </p>
          </div>
        </div>

        <div className="flex items-center bg-[#16181c] border border-[#252830] rounded-xl px-3.5 h-10 gap-2.5 focus-within:border-amber-500/40 transition-colors">
          <Search className="w-4 h-4 text-zinc-500" />
          <input
            type="text"
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder="Search bookmarks..."
            className="flex-1 bg-transparent text-sm text-zinc-100 outline-none placeholder-zinc-500"
          />
          {query && (
            <button onClick={() => setQuery('')} className="text-zinc-500 hover:text-zinc-300">
              <X className="w-3.5 h-3.5" />
            </button>
          )}
        </div>

        {grouped.length === 0 && (
          <div className="text-center py-16 text-zinc-500 text-sm">
            {query ? 'No matching bookmarks.' : 'No bookmarks yet — click the star in the address bar to save a page.'}
          </div>
        )}

        <div className="space-y-6">
          {grouped.map(([folder, entries]) => (
            <div key={folder} className="space-y-1.5">
              <h2 className="flex items-center gap-1.5 text-xs font-bold uppercase text-zinc-500 tracking-wider px-1">
                <Folder className="w-3 h-3" />
                {folder}
              </h2>
              <div className="bg-[#16181c] border border-[#252830] rounded-xl divide-y divide-[#252830] overflow-hidden">
                {entries.map((item) => (
                  <div
                    key={item.id}
                    className="group flex items-center gap-3 px-4 py-2.5 hover:bg-[#1f2228] transition-colors cursor-pointer"
                    onClick={() => openUrl(item.url, item.title)}
                  >
                    <div className="flex-1 min-w-0">
                      <div className="text-sm font-medium text-zinc-100 truncate">{item.title}</div>
                      <div className="text-xs font-mono text-zinc-500 truncate">{item.url}</div>
                    </div>
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        removeBookmark(item.id);
                      }}
                      className="opacity-0 group-hover:opacity-100 p-1.5 text-zinc-500 hover:text-red-400 hover:bg-red-500/10 rounded-md transition-all flex-shrink-0"
                      title="Remove bookmark"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>
                ))}
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
